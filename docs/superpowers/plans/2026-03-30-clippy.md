# Clippy Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a local-network clipboard sharing service — named rooms, syntax-highlighted snippets, at-rest encryption, auto-expiry, deployed as a single Docker Compose service.

**Architecture:** Single Go binary. HTTP handlers render Go templates (full pages + HTMX fragments). SQLite stores rooms and snippets via `modernc.org/sqlite` (pure Go, no CGO). AES-256-GCM encrypts snippet content at rest. A background goroutine purges expired snippets hourly.

**Tech Stack:** Go 1.25, `modernc.org/sqlite`, `golang.org/x/crypto` (HKDF), HTMX 2.x, highlight.js 11.x, Docker Compose.

```go
// internal/config/config.go
package config

import (
	"errors"
	"os"
	"strconv"
)

type Config struct {
	Secret          string
	Port            string
	DefaultTTLHours int
	DBPath          string
}

func Load() (*Config, error) {
	secret := os.Getenv("CLIPPY_SECRET")
	if secret == "" {
		return nil, errors.New("CLIPPY_SECRET is required")
	}
	if len(secret) < 32 {
		return nil, errors.New("CLIPPY_SECRET must be at least 32 characters")
	}

	port := os.Getenv("CLIPPY_PORT")
	if port == "" {
		port = "8080"
	}

	ttl := 24
	if v := os.Getenv("CLIPPY_DEFAULT_TTL_HOURS"); v != "" {
		n, err := strconv.Atoi(v)
		if err != nil || n <= 0 {
			return nil, errors.New("CLIPPY_DEFAULT_TTL_HOURS must be a positive integer")
		}
		ttl = n
	}

	dbPath := os.Getenv("CLIPPY_DB_PATH")
	if dbPath == "" {
		dbPath = "/data/clippy.db"
	}

	return &Config{
		Secret:          secret,
		Port:            port,
		DefaultTTLHours: ttl,
		DBPath:          dbPath,
	}, nil
}
```

```go
// internal/model/model.go
package model

import (
	"math/rand/v2"
	"sort"
	"strings"
	"time"
)

const idAlphabet = "abcdefghijklmnopqrstuvwxyz0123456789"

type Room struct {
	ID        string
	CreatedAt time.Time
	PinHash   string // empty in phase 1
}

type Snippet struct {
	ID        string
	RoomID    string
	Name      string
	Content   string // decrypted; never persisted directly
	Language  string
	Tags      []string
	ExpiresAt time.Time
	CreatedAt time.Time
	UpdatedAt time.Time
}

func GenerateID() string {
	b := make([]byte, 6)
	for i := range b {
		b[i] = idAlphabet[rand.IntN(len(idAlphabet))]
	}
	return string(b)
}

func ParseTags(s string) []string {
	s = strings.TrimSpace(s)
	if s == "" {
		return []string{}
	}
	parts := strings.Split(s, ",")
	out := make([]string, 0, len(parts))
	for _, p := range parts {
		p = strings.TrimSpace(p)
		if p != "" {
			out = append(out, p)
		}
	}
	return out
}

func JoinTags(tags []string) string {
	return strings.Join(tags, ",")
}

func MatchesAllTags(snippetTags, filterTags []string) bool {
	if len(filterTags) == 0 {
		return true
	}
	set := make(map[string]bool, len(snippetTags))
	for _, t := range snippetTags {
		set[t] = true
	}
	for _, t := range filterTags {
		if !set[t] {
			return false
		}
	}
	return true
}

func UniqueTags(snippets []Snippet) []string {
	set := make(map[string]bool)
	for _, s := range snippets {
		for _, t := range s.Tags {
			set[t] = true
		}
	}
	out := make([]string, 0, len(set))
	for t := range set {
		out = append(out, t)
	}
	sort.Strings(out)
	return out
}
```

```go
// internal/crypto/crypto.go
package crypto

import (
	"crypto/aes"
	"crypto/cipher"
	"crypto/rand"
	"crypto/sha256"
	"errors"
	"io"

	"golang.org/x/crypto/hkdf"
)

// DeriveKey derives a 32-byte AES key from the secret using HKDF-SHA256.
// Deterministic: same secret always produces the same key.
func DeriveKey(secret []byte) ([]byte, error) {
	r := hkdf.New(sha256.New, secret, nil, []byte("clippy-v1-snippet-content"))
	key := make([]byte, 32)
	if _, err := io.ReadFull(r, key); err != nil {
		return nil, err
	}
	return key, nil
}

// Encrypt encrypts plaintext with AES-256-GCM. Returns nonce || ciphertext.
func Encrypt(key, plaintext []byte) ([]byte, error) {
	block, err := aes.NewCipher(key)
	if err != nil {
		return nil, err
	}
	gcm, err := cipher.NewGCM(block)
	if err != nil {
		return nil, err
	}
	nonce := make([]byte, gcm.NonceSize())
	if _, err = rand.Read(nonce); err != nil {
		return nil, err
	}
	// Seal appends ciphertext to nonce, giving us nonce || ciphertext
	return gcm.Seal(nonce, nonce, plaintext, nil), nil
}

// Decrypt decrypts data produced by Encrypt. Returns an error if authentication fails.
func Decrypt(key, data []byte) ([]byte, error) {
	block, err := aes.NewCipher(key)
	if err != nil {
		return nil, err
	}
	gcm, err := cipher.NewGCM(block)
	if err != nil {
		return nil, err
	}
	nonceSize := gcm.NonceSize()
	if len(data) < nonceSize {
		return nil, errors.New("ciphertext too short")
	}
	nonce, ciphertext := data[:nonceSize], data[nonceSize:]
	return gcm.Open(nil, nonce, ciphertext, nil)
}
```

```go
// internal/db/db.go
package db

import (
	"database/sql"
	"errors"

	_ "modernc.org/sqlite"
)

var ErrNotFound = errors.New("not found")

type DB struct {
	sql *sql.DB
}

func Open(path string) (*DB, error) {
	conn, err := sql.Open("sqlite", path)
	if err != nil {
		return nil, err
	}
	conn.SetMaxOpenConns(1) // SQLite is single-writer
	database := &DB{sql: conn}
	if err := database.migrate(); err != nil {
		conn.Close()
		return nil, err
	}
	return database, nil
}

func (d *DB) Close() error {
	return d.sql.Close()
}

func (d *DB) migrate() error {
	_, err := d.sql.Exec(`
		PRAGMA journal_mode=WAL;
		PRAGMA foreign_keys=ON;

		CREATE TABLE IF NOT EXISTS rooms (
			id         TEXT PRIMARY KEY,
			created_at INTEGER NOT NULL,
			pin_hash   TEXT
		);

		CREATE TABLE IF NOT EXISTS snippets (
			id          TEXT PRIMARY KEY,
			room_id     TEXT NOT NULL REFERENCES rooms(id) ON DELETE CASCADE,
			name        TEXT,
			content_enc BLOB NOT NULL,
			language    TEXT NOT NULL DEFAULT 'plaintext',
			tags        TEXT NOT NULL DEFAULT '',
			expires_at  INTEGER NOT NULL,
			created_at  INTEGER NOT NULL,
			updated_at  INTEGER NOT NULL
		);

		CREATE INDEX IF NOT EXISTS idx_snippets_room_expires
			ON snippets(room_id, expires_at);
	`)
	return err
}
```

```go
// internal/db/rooms.go
package db

import (
	"database/sql"
	"time"

	"clippy/internal/model"
)

func (d *DB) CreateRoom(r *model.Room) error {
	_, err := d.sql.Exec(
		`INSERT INTO rooms (id, created_at, pin_hash) VALUES (?, ?, ?)`,
		r.ID, r.CreatedAt.Unix(), r.PinHash,
	)
	return err
}

func (d *DB) GetRoom(id string) (*model.Room, error) {
	row := d.sql.QueryRow(`SELECT id, created_at, pin_hash FROM rooms WHERE id = ?`, id)
	var r model.Room
	var createdAt string
	var pinHash sql.NullString
	if err := row.Scan(&r.ID, &createdAt, &pinHash); err != nil {
		if err == sql.ErrNoRows {
			return nil, ErrNotFound
		}
		return nil, err
	}
	r.CreatedAt = time.Unix(createdAt, 0).UTC()
	r.PinHash = pinHash.String
	return &r, nil
}

func (d *DB) ListRooms() ([]*model.Room, error) {
	rows, err := d.sql.Query(`SELECT id, created_at, pin_hash FROM rooms ORDER BY created_at`)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var rooms []*model.Room
	for rows.Next() {
		var r model.Room
		var createdAt string
		var pinHash sql.NullString
		if err := rows.Scan(&r.ID, &createdAt, &pinHash); err != nil {
			return nil, err
		}
		r.CreatedAt = time.Unix(createdAt, 0).UTC()
		r.PinHash = pinHash.String
		rooms = append(rooms, &r)
	}
	return rooms, rows.Err()
}

func (d *DB) DeleteRoom(id string) error {
	_, err := d.sql.Exec(`DELETE FROM rooms WHERE id = ?`, id)
	return err
}
```

```go
package db

import (
	"database/sql"

	"clippy/internal/model"

func (d *DB) CreateSnippet(s *model.DBSnippet) error {
	_, err := d.sql.Exec(`
		INSERT INTO snippets (id, room_id, name, content_enc, language, tags, expires_at, created_at, updated_at)
		VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
		s.ID, s.RoomID, s.Name, s.ContentEnc, s.Language, s.Tags,
		s.ExpiresAt.UTC(), s.CreatedAt.UTC(), s.UpdatedAt.UTC(),
	)
	return err
}

func (d *DB) GetSnippet(id string) (*model.DBSnippet, error) {
	row := d.sql.QueryRow(`
		SELECT id, room_id, name, content_enc, language, tags, expires_at, created_at, updated_at
		FROM snippets WHERE id = ?`, id)
	return scanSnippet(row)
}

func (d *DB) ListSnippets(roomID string) ([]*model.DBSnippet, error) {
	rows, err := d.sql.Query(`
		SELECT id, room_id, name, content_enc, language, tags, expires_at, created_at, updated_at
		FROM snippets
		WHERE room_id = ? AND expires_at > unixepoch()
		ORDER BY created_at DESC`, roomID)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var snippets []*model.DBSnippet
	for rows.Next() {
		s, err := scanSnippet(rows)
		if err != nil {
			return nil, err
		}
		snippets = append(snippets, s)
	}
	return snippets, rows.Err()
}

func (d *DB) UpdateSnippet(s *model.DBSnippet) error {
	_, err := d.sql.Exec(`
		UPDATE snippets SET name=?, content_enc=?, language=?, tags=?, expires_at=?, updated_at=?
		WHERE id=?`,
		s.Name, s.ContentEnc, s.Language, s.Tags,
		s.ExpiresAt.UTC(), s.UpdatedAt.UTC(), s.ID,
	)
	return err
}

func (d *DB) DeleteExpired() (int64, error) {
	res, err := d.sql.Exec(`DELETE FROM snippets WHERE expires_at < unixepoch()`)
	if err != nil {
		return 0, err
	}
	return res.RowsAffected()
}

type scanner interface {
	Scan(dest ...any) error
}

func scanSnippet(s scanner) (*model.DBSnippet, error) {
	var snap model.DBSnippet
	var name sql.NullString
	var expiresAt, createdAt, updatedAt string
	err := s.Scan(&snap.ID, &snap.RoomID, &name, &snap.ContentEnc,
		&snap.Language, &snap.Tags, &expiresAt, &createdAt, &updatedAt)
	if err != nil {
		if err == sql.ErrNoRows {
			return nil, ErrNotFound
		}
		return nil, err
	}
	snap.Name = name.String
	snap.ExpiresAt = time.Unix(expiresAt, 0).UTC()
	snap.CreatedAt = time.Unix(createdAt, 0).UTC()
	snap.UpdatedAt = time.Unix(updatedAt, 0).UTC()
	return &snap, nil
}
```

```go
// internal/worker/expiry.go
package worker

import (
	"context"
	"log"
	"time"

	"clippy/internal/db"
)

// Start runs a background goroutine that deletes expired snippets on the given interval.
// It stops when ctx is cancelled.
func Start(ctx context.Context, database *db.DB, interval time.Duration) {
	go func() {
		ticker := time.NewTicker(interval)
		defer ticker.Stop()
		for {
			select {
			case <-ctx.Done():
				return
			case <-ticker.C:
				n, err := database.DeleteExpired()
				if err != nil {
					log.Printf("expiry worker: %v", err)
				} else if n > 0 {
					log.Printf("expiry worker: deleted %d expired snippets", n)
				}
			}
		}
	}()
}
```

```go
package handler

import (
```

```go
	"html/template"
	"net/http"
	"net/url"

	"strings"
	"time"
)

var slugRe = regexp.MustCompile(`^[a-z0-9][a-z0-9\-]{0,49}$`)

func validateSlug(s string) bool {
	return slugRe.MatchString(s) && !strings.HasPrefix(s, "_")
}

func isHTMX(r *http.Request) bool {
	return r.Header.Get("HX-Request") == "true"
}

func (h *Handler) render(w http.ResponseWriter, fullTmpl, fragTmpl string, data any) {
	// caller decides which template name to use
	_ = fullTmpl
	_ = fragTmpl
	// This overload is unused — use renderFull/renderFrag directly.
}

func (h *Handler) renderFull(w http.ResponseWriter, name string, data any) {
	w.Header().Set("Content-Type", "text/html; charset=utf-8")
	if err := h.tmpl.ExecuteTemplate(w, name, data); err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
	}
}

func (h *Handler) renderFrag(w http.ResponseWriter, name string, data any) {
	w.Header().Set("Content-Type", "text/html; charset=utf-8")
	if err := h.tmpl.ExecuteTemplate(w, name, data); err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
	}
}

func (h *Handler) renderAuto(w http.ResponseWriter, r *http.Request, fullTmpl, fragTmpl string, data any) {
	if isHTMX(r) {
		h.renderFrag(w, fragTmpl, data)
	} else {
		h.renderFull(w, fullTmpl, data)
	}
}

func expiryDisplay(t time.Time) string {
	d := time.Until(t)
	if d <= 0 {
		return "expired"
	}
	if d < time.Hour {
		return fmt.Sprintf("%dm", int(d.Minutes()))
	}
	if d < 24*time.Hour {
		return fmt.Sprintf("%dh", int(d.Hours()))
	}
	return fmt.Sprintf("%dd", int(d.Hours()/24))
}

func expiringSoon(t time.Time) bool {
	return time.Until(t) < time.Hour
}

// TemplateFuncMap returns the functions needed by templates.
func TemplateFuncMap() template.FuncMap {
	return template.FuncMap{
		"truncate": func(s string, n int) string {
			// Get first line, truncate to n chars
			line := strings.SplitN(s, "\n", 2)[0]
			if len(line) > n {
				return line[:n] + "…"
			}
			return line
		},
		"isActive": func(tag string, active []string) bool {
			for _, a := range active {
				if a == tag {
					return true
				}
			}
			return false
		},
		"tagQuery": func(tag string, active []string) string {
			// Toggle tag in/out of active set and return query string
			next := make([]string, 0, len(active)+1)
			found := false
			for _, a := range active {
				if a == tag {
					found = true
				} else {
					next = append(next, a)
				}
			}
			if !found {
				next = append(next, tag)
			}
			vals := url.Values{}
			for _, t := range next {
				vals.Add("tag", t)
			}
			return vals.Encode()
		},
		"joinTags": func(tags []string) string {
			return strings.Join(tags, ", ")
		},
	}
}
```

```go
func (h *Handler) createRoom(w http.ResponseWriter, r *http.Request) {
	if err := r.ParseForm(); err != nil {
		http.Error(w, "bad request", http.StatusBadRequest)
		return
	}
	id := r.FormValue("id")
	if !validateSlug(id) {
		http.Error(w, "invalid room name: use lowercase letters, numbers, hyphens only", http.StatusBadRequest)

		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}
	http.Redirect(w, r, "/"+id, http.StatusSeeOther)
}
```

```go
// internal/handler/room_view.go
package handler

import (
	"net/http"

	"clippy/internal/db"
	"clippy/internal/model"
)

func (h *Handler) roomView(w http.ResponseWriter, r *http.Request) {
	roomID := r.PathValue("room")

	// Validate room exists
	if _, err := h.db.GetRoom(roomID); err == db.ErrNotFound {
		http.NotFound(w, r)
		return
	} else if err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}

	// Parse tag filters from query string
	activeTags := r.URL.Query()["tag"]

	// Load all non-expired snippets for this room
	dbSnippets, err := h.db.ListSnippets(roomID)
	if err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}

	// Decrypt and build view models
	allSnippets := make([]model.Snippet, 0, len(dbSnippets))
	for _, ds := range dbSnippets {
		content, err := h.decrypt(ds.ContentEnc)
		if err != nil {
			http.Error(w, "decryption failed", http.StatusInternalServerError)
			return
		}
		allSnippets = append(allSnippets, model.Snippet{
			ID:        ds.ID,
			RoomID:    ds.RoomID,
			Name:      ds.Name,
			Content:   content,
			Language:  ds.Language,
			Tags:      model.ParseTags(ds.Tags),
			ExpiresAt: ds.ExpiresAt,
			CreatedAt: ds.CreatedAt,
			UpdatedAt: ds.UpdatedAt,
		})
	}

	allTags := model.UniqueTags(allSnippets)

	// Filter in Go if tags are selected
	filtered := make([]SnippetView, 0, len(allSnippets))
	for _, s := range allSnippets {
		if model.MatchesAllTags(s.Tags, activeTags) {
			filtered = append(filtered, SnippetView{
				Snippet:       s,
				ExpiryDisplay: expiryDisplay(s.ExpiresAt),
				ExpiringSoon:  expiringSoon(s.ExpiresAt),
			})
		}
	}

	data := RoomViewData{
		RoomID:     roomID,
		Snippets:   filtered,
		AllTags:    allTags,
		ActiveTags: activeTags,
		Tags:       len(allTags) > 0,
	}

	h.renderAuto(w, r, "room-page", "snippet-table", data)
}

func (h *Handler) snippetForm(w http.ResponseWriter, r *http.Request) {
	roomID := r.PathValue("room")
	h.renderFrag(w, "snippet-form", map[string]string{"RoomID": roomID})
}
```

```go
package handler

import (

func (h *Handler) createSnippet(w http.ResponseWriter, r *http.Request) {
	roomID := r.PathValue("room")

	if _, err := h.db.GetRoom(roomID); err == db.ErrNotFound {
		http.NotFound(w, r)
		return
	}

	if err := r.ParseForm(); err != nil {
		http.Error(w, "bad request", http.StatusBadRequest)
		return
	}

	content := r.FormValue("content")
	if content == "" {
		http.Error(w, "content is required", http.StatusBadRequest)
		return
	}

	ttlHours := h.cfg.DefaultTTLHours
	if v := r.FormValue("ttl_hours"); v != "" {
		if n, err := strconv.Atoi(v); err == nil && n > 0 {
			ttlHours = n
		}
	}

	enc, err := h.encrypt(content)
	if err != nil {
		http.Error(w, "encryption failed", http.StatusInternalServerError)
		return
	}

	now := time.Now()
	s := &model.DBSnippet{
		ID:         model.GenerateID(),
		RoomID:     roomID,
		Name:       r.FormValue("name"),
		ContentEnc: enc,
		Language:   r.FormValue("language"),
		Tags:       model.JoinTags(model.ParseTags(r.FormValue("tags"))),
		ExpiresAt:  now.Add(time.Duration(ttlHours) * time.Hour),
		CreatedAt:  now,
		UpdatedAt:  now,
	}
	if s.Language == "" {
		s.Language = "plaintext"
	}

	if err := h.db.CreateSnippet(s); err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}

	http.Redirect(w, r, "/"+roomID+"/"+s.ID, http.StatusSeeOther)
}

func (h *Handler) viewSnippet(w http.ResponseWriter, r *http.Request) {
	roomID := r.PathValue("room")
	id := r.PathValue("id")

			UpdatedAt: ds.UpdatedAt,
		},
		ExpiryDisplay: expiryDisplay(ds.ExpiresAt),
		ExpiringSoon:  expiringSoon(ds.ExpiresAt),
	}

	data := SnippetPageData{
		RoomID:    roomID,
		Snippet:   sv,
		Languages: languages,
	}

	h.renderAuto(w, r, "snippet-page", "snippet-view", data)
}

func (h *Handler) updateSnippet(w http.ResponseWriter, r *http.Request) {
	roomID := r.PathValue("room")
	id := r.PathValue("id")

	if err == db.ErrNotFound {
		http.NotFound(w, r)

		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}

	if err := r.ParseForm(); err != nil {
		http.Error(w, "bad request", http.StatusBadRequest)
		return
	}

	content := r.FormValue("content")

	ttlHours := h.cfg.DefaultTTLHours
	if v := r.FormValue("ttl_hours"); v != "" {

		}
	}

	enc, err := h.encrypt(content)
	if err != nil {
		http.Error(w, "encryption failed", http.StatusInternalServerError)
		return
	}

	ds.Name = r.FormValue("name")
	ds.ContentEnc = enc
	ds.Language = r.FormValue("language")
	if ds.Language == "" {
		ds.Language = "plaintext"

	ds.Tags = model.JoinTags(model.ParseTags(r.FormValue("tags")))

		http.Error(w, err.Error(), http.StatusInternalServerError)
		return

	// Re-fetch to get fresh data
	r2 := r.Clone(r.Context())
	r2.Method = "GET"
	h.viewSnippet(w, r2)
}

		return

		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}

	content, err := h.decrypt(ds.ContentEnc)
	if err != nil {
		http.Error(w, "decryption failed", http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "text/plain; charset=utf-8")
	w.Write([]byte(content))
}
```

```go
// cmd/clippy/main.go
package main

import (
	"context"
	"html/template"
	"log"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	"clippy/internal/config"
	"clippy/internal/crypto"
	"clippy/internal/db"
	"clippy/internal/handler"
	"clippy/internal/worker"
)

func main() {
	cfg, err := config.Load()
	if err != nil {
		log.Fatalf("config: %v", err)
	}

	key, err := crypto.DeriveKey([]byte(cfg.Secret))
	if err != nil {
		log.Fatalf("crypto: %v", err)
	}

	database, err := db.Open(cfg.DBPath)
	if err != nil {
		log.Fatalf("db: %v", err)
	}
	defer database.Close()

	tmpl, err := template.New("").
		Funcs(handler.TemplateFuncMap()).
		ParseFS(templateFS, "templates/*.html", "templates/fragments/*.html")
	if err != nil {
		log.Fatalf("templates: %v", err)
	}

	h := handler.New(database, key, tmpl, cfg)
	routes := h.Routes(staticHTTPFS())

	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer stop()

	worker.Start(ctx, database, time.Hour)

	srv := &http.Server{
		Addr:    ":" + cfg.Port,
		Handler: routes,
	}

	log.Printf("clippy listening on :%s", cfg.Port)
	go func() {
		if err := srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			log.Fatalf("server: %v", err)
		}
	}()

	<-ctx.Done()
	log.Println("shutting down...")
	shutCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	srv.Shutdown(shutCtx)
}
```
